// 1. fetch(주소): 서버에 "이 주소로 데이터를 주세요" 하고 요청을 보냅니다.
fetch("http://localhost:10000/reservations")

    // 2. 첫 번째 .then(res): 서버가 요청을 받고 '응답 봉투(res)'를 보내왔습니다.
    // res는 Response(응답)의 줄임말입니다.
    .then(res => {
        // res.ok는 서버가 200 OK 응답을 주었는지 확인하는 안전장치입니다.
        if (!res.ok) {
            throw new Error("서버 응답 실패!");
        }
        
        // res.json(): 봉투 안에 담긴 문자열 데이터를 자바스크립트가 읽을 수 있는 객체/배열로 바꿔줍니다.
        // 다음 .then으로 바꿔진 데이터(Promise)를 넘겨줍니다.
        return res.json(); 
    })

    // 3. 두 번째 .then(data): 위에서 res.json()으로 완벽히 해독된 데이터가 'data'에 들어옵니다.
    // data는 이름일 뿐이라 원하는 변수명(예: result, reservations)으로 바꿔도 됩니다.
    .then(data => {
        // 이제 data를 가지고 화면에 출력하거나 원하는 작업을 합니다!
        console.log("서버에서 받은 실제 데이터:", data);
    })

    // 4. .catch(error): 요청하는 도중 인터넷 연결이 끊기거나 서버 에러가 나면 실행됩니다.
    .catch(error => {
        console.error("에러 발생:", error);
    });

// 서버로 보낼 데이터 준비
const newPet = { petName: "초코", ownerName: "홍길동" };

// 1. fetch(주소, 옵션): 두 번째 파라미터로 전송 방식과 보낼 내용을 지정합니다.
fetch("http://localhost:10000/reservations", {
    method: "POST", // HTTP 요청 메서드 (POST: 생성, PUT: 수정, DELETE: 삭제)
    headers: {
        "Content-Type": "application/json" // "내가 지금 보내는 데이터는 JSON 형태입니다"라고 서버에 알려줌
    },
    body: JSON.stringify(newPet) // 자바스크립트 객체(newPet)를 서버로 전송하기 위해 문자열(JSON)로 변환!
})

    // 2. 서버가 등록을 잘 완료했다는 응답(res)을 보내옴
    .then(res => {
        // 성공했는지 확인
        if (res.ok) {
            alert("성공적으로 등록되었습니다!");
            return res.json(); // 서버가 등록 결과 데이터를 다시 반환해 준다면 해독함
        }
    })

    // 3. 등록 후 서버가 보내준 결과 데이터
    .then(data => {
        console.log("등록 완료된 데이터 정보:", data);
    })

    // 4. 에러 처리
    .catch(error => {
        console.error("등록 실패:", error);
    });

// ==============================================================================
// 1. 전역 변수 및 상수 선언 (글로벌 세팅)
// ==============================================================================

// [개념] API 주소 상수(const) 설정
// 백엔드 스프링 부트 서버의 URL입니다. 나중에 주소가 바뀌어도 이 변수만 수정하면 되므로 관리가 용이합니다.
const API_URL = "http://localhost:10000/reservations"; 

// [개념] DOM (Document Object Model) 요소 가져오기
// document.querySelector는 HTML의 id(#)나 클래스(.)에 접근하여 JS 변수에 저장(참조)하는 명령입니다.
// 이렇게 저장해 두어야 나중에 JS에서 입력된 값(.value)을 읽거나 글자(.textContent)를 바꿀 수 있습니다.

const reservationId = document.querySelector("#reservationId"); // 수정할 때 필요한 예약 ID (hidden input)
const petName = document.querySelector("#petName");             // 반려동물 이름 입력창
const ownerName = document.querySelector("#ownerName");         // 보호자 이름 입력창
const animalType = document.querySelector("#animalType");       // 동물 종류 선택창 (select)
const doctorName = document.querySelector("#doctorName");       // 담당 수의사 입력창
const reservationDate = document.querySelector("#reservationDate"); // 예약 날짜 선택창
const status = document.querySelector("#status");               // 예약 상태 선택창
const price = document.querySelector("#price");                 // 예상 진료비 입력창

const saveBtn = document.querySelector("#saveBtn");             // '등록/수정' 버튼
const cancelBtn = document.querySelector("#cancelBtn");         // '취소' 버튼
const reservationList = document.querySelector("#reservationList"); // 목록이 그려질 <tbody> 영역


// ==============================================================================
// 2. 전체 예약 목록 조회 함수 (GET 요청)
// ==============================================================================

// [개념] 화살표 함수 () => { ... }
// loadReservations라는 이름의 함수를 정의합니다. (매개변수/입력값이 없으므로 빈 괄호 () 사용)
const loadReservations = () => {
    
    // [개념] HTML 화면 초기화
    // 데이터를 다시 새로고침해서 그릴 때, 기존 목록이 누적해서 덧붙여지는 것을 방지하기 위해 싹 비웁니다.
    reservationList.innerHTML = "";

    // [개념] fetch(URL) - GET 요청
    // 기본적으로 옵션 없이 fetch(URL)만 쓰면 백엔드에 "데이터 조회(GET)해 주세요"라는 요청을 보냅니다.
    fetch(API_URL)
    
        // [개념] 첫 번째 .then(res => res.json())
        // res(응답 객체)를 받아 해석 가능한 JSON 형식으로 해독(파싱)하여 다음 .then으로 전달합니다.
        .then(res => res.json())
        
        // [개념] 두 번째 .then(data => { ... })
        // data에는 해독 완료된 실제 백엔드 데이터(예약 객체 배열 [...])가 들어옵니다.
        .then(data => {
            
            // [개념] data.forEach(...)
            // 배열 안의 데이터 개수만큼 반복하며 하나씩 reservation이라는 변수로 꺼냅니다.
            data.forEach(reservation => {
                
                // [개념] document.createElement("tr")
                // 메모리상에 새로운 <tr>(테이블 행) HTML 태그 요소를 하나 생성합니다.
                const tr = document.createElement("tr");

                // [개념] 템플릿 리터럴 백틱(`)과 ${} 
                // 백틱(`)을 사용하면 문자열 사이에 변수(${reservation.petName})를 편하게 삽입할 수 있습니다.
                // toLocaleString()은 숫자 3자리마다 쉼표(,)를 찍어주는 자바스크립트 내장 함수입니다.
                tr.innerHTML = `
                    <td>${reservation.id}</td>
                    <td>${reservation.petName}</td>
                    <td>${reservation.ownerName}</td>
                    <td>${reservation.animalType}</td>
                    <td>${reservation.doctorName}</td>
                    <td>${reservation.reservationDate}</td>
                    <td>${reservation.status}</td>
                    <td>${reservation.price.toLocaleString()}원</td>
                    <td>
                        <!-- 버튼 클릭 시 해당 예약 id를 매개변수로 넘겨주는 함수 호출 -->
                        <button class="edit-btn" onclick="editReservation(${reservation.id})">수정</button>
                        <button class="delete-btn" onclick="deleteReservation(${reservation.id})">삭제</button>
                    </td>
                `;

                // [개념] appendChild()
                // 완성된 <tr> 조각을 실제 HTML 화면의 <tbody>(reservationList) 안의 자식으로 붙여서 표시합니다.
                reservationList.appendChild(tr);
            });
        })
        
        // [개념] .catch(error => { ... })
        // 통신 중 서버가 꺼져 있거나 주소가 틀리는 등 에러가 발생하면 이 블록이 실행됩니다.
        .catch(error => {
            console.error(error);
            alert("예약 정보를 불러올 수 없습니다.");
        });
};


// ==============================================================================
// 3. 등록 / 수정 버튼 클릭 이벤트 (유효성 검사 및 분기)
// ==============================================================================

// [개념] addEventListener("click", () => { ... })
// saveBtn(등록 버튼)을 마우스로 클릭하는 순간 실행될 동작(이벤트 리스너)을 등록합니다.
saveBtn.addEventListener("click", () => {

    // [개념] 유효성 검사 (Validation) & .trim() & .focus()
    // trim()은 입력값 양쪽의 불필요한 공백을 제거합니다.
    // 입력값이 비어있다면 alert 띄우고, focus()로 해당 입력창에 커서를 위치시킨 뒤, return으로 함수를 종료합니다.
    if (petName.value.trim() === "") {
        alert("반려동물 이름을 입력하세요");
        petName.focus();
        return; // 이후 아래 코드가 실행되지 않고 빠져나감
    }

    if (ownerName.value.trim() === "") {
        alert("보호자 이름을 입력하세요");
        ownerName.focus();
        return;
    }

    if (animalType.value === "") {
        alert("동물 종류를 선택하세요");
        animalType.focus();
        return;
    }

    if (doctorName.value.trim() === "") {
        alert("담당 수의사 이름을 입력하세요");
        doctorName.focus();
        return;
    }

    if (reservationDate.value === "") {
        alert("예약 날짜를 선택하세요");
        reservationDate.focus();
        return;
    }

    if (price.value === "") {
        alert("예상 진료비를 작성하세요");
        price.focus();
        return;
    }

    // [개념] 자바스크립트 객체(Object) 생성
    // 서버로 보낼 데이터를 Key: Value 형태로 바인딩합니다.
    // price는 숫자 타입이어야 하므로 parseInt()를 사용해 문자열을 정수(Integer)로 변환합니다.
    const reservation = {
        petName: petName.value,
        ownerName: ownerName.value,
        animalType: animalType.value,
        doctorName: doctorName.value,
        reservationDate: reservationDate.value,
        status: status.value,
        price: parseInt(price.value)
    };

    // [개념] 삼항 판단 및 모드 분기 (등록 vs 수정)
    // reservationId(hidden input)에 번호 값이 들어있는지 체크합니다.
    // id가 비어있으면 "신규 등록(POST)" 상태이고, id가 들어있으면 "기존 데이터 수정(PUT)" 상태입니다.
    const id = reservationId.value;
    if (id) {
        updateReservation(id, reservation); // 수정 함수 호출
    } else {
        insertReservation(reservation);     // 신규 등록 함수 호출
    }
});


// ==============================================================================
// 4. 신규 데이터 저장 처리 함수 (POST 요청)
// ==============================================================================

// [개념] 매개변수가 있는 함수 (reservation이라는 객체를 전달받음)
const insertReservation = (reservation) => {
    
    // [개념] fetch(URL, { 옵션 객체 })
    // 서버로 데이터 전달 시 method, headers, body 옵션을 지정합니다.
    fetch(API_URL, {
        method: "POST", // 백엔드에 "새 데이터를 추가해라"는 요청
        headers: { 
            "Content-Type": "application/json" // "내가 보내는 데이터 형식은 JSON이다"라고 알려주는 통신 규약
        },
        // [개념] JSON.stringify()
        // 자바스크립트 객체(reservation)를 네트워크 전송이 가능한 'JSON 문자열' 형태로 변환(직렬화)합니다.
        body: JSON.stringify(reservation) 
    })
    .then(res => {
        // [개념] res.ok
        // 서버의 응답 상태 코드가 200~299 범위(성공)이면 true를 반환합니다.
        if (res.ok) {
            alert("예약이 등록되었습니다.");
            resetForm();        // 입력 폼 비우기
            loadReservations(); // 목록 새로고침 (DB 변경사항 반영)
        } else {
            alert("등록 실패");
        }
    })
    .catch(error => console.error("등록 오류:", error));
};


// ==============================================================================
// 5. 수정할 데이터 조회 및 입력창 채우기 함수 (GET 단건 조회)
// ==============================================================================

const editReservation = (id) => {
    // [개념] URL Path Variable 경로 (예: http://localhost:10000/reservations/3)
    // 백엔드 단건 조회 API 주소 형태로 id를 뒤에 붙여 요청합니다.
    fetch(`${API_URL}/${id}`)
        .then(res => res.json())
        .then(data => {
            // 서버에서 받아온 기존 예약 정보를 각각의 입력창 HTML 요쇼에 넣어줍니다.
            reservationId.value = data.id; // 핵심! id를 hidden 필드에 숨겨두어 수정 모드로 전환
            petName.value = data.petName;
            ownerName.value = data.ownerName;
            animalType.value = data.animalType;
            doctorName.value = data.doctorName;
            reservationDate.value = data.reservationDate;
            status.value = data.status;
            price.value = data.price;

            // [개념] 버튼 텍스트 변경
            // 사용자에게 지금 상태가 '수정' 모드임을 알리기 위해 버튼 글자를 변경합니다.
            saveBtn.textContent = "수정 완료"; 
        })
        .catch(error => console.error("조회 오류:", error));
};


// ==============================================================================
// 6. 데이터 수정 저장 처리 함수 (PUT 요청)
// ==============================================================================

const updateReservation = (id, reservation) => {
    // [개념] HTTP PUT 요청
    // 기존에 존재하는 데이터 id의 전체 정보를 수정할 때 사용하는 HTTP 메서드입니다.
    fetch(`${API_URL}/${id}`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(reservation)
    })
    .then(res => {
        if (res.ok) {
            alert("예약이 수정되었습니다.");
            resetForm();        // 폼 초기화 및 버튼 글자 원복
            loadReservations(); // 수정된 내용으로 테이블 재조회
        } else {
            alert("수정 실패");
        }
    })
    .catch(error => console.error("수정 오류:", error));
};


// ==============================================================================
// 7. 데이터 삭제 처리 함수 (DELETE 요청)
// ==============================================================================

const deleteReservation = (id) => {
    // [개념] confirm() 창
    // 확인(true) / 취소(false)를 선택할 수 있는 팝업창입니다.
    // !confirm(...)은 사용자가 '취소'를 누르면 return하여 삭제 작업을 즉시 중단시킵니다.
    if (!confirm(`${id}번 예약을 정말 삭제하시겠습니까?`)) return;

    // [개념] HTTP DELETE 요청
    // 해당 ID의 데이터를 삭제하라고 백엔드 서버에 전달합니다. (보낼 body 데이터는 없음)
    fetch(`${API_URL}/${id}`, { method: "DELETE" })
        .then(res => {
            if (res.ok) {
                alert("삭제되었습니다.");
                loadReservations(); // 삭제된 후의 목록을 다시 불러옴
            } else {
                alert("삭제 실패");
            }
        })
        .catch(error => console.error("삭제 오류:", error));
};


// ==============================================================================
// 8. 입력 폼 초기화 헬퍼 함수
// ==============================================================================

const resetForm = () => {
    // 모든 input과 select의 값(.value)을 빈 값으로 리셋합니다.
    reservationId.value = "";
    petName.value = "";
    ownerName.value = "";
    animalType.value = "";
    doctorName.value = "";
    reservationDate.value = "";
    status.value = "예약";
    price.value = "";
    
    // 수정 모드에서 바뀌었던 버튼 글자를 다시 원래대로 '등록'으로 되돌려 놓습니다.
    saveBtn.textContent = "등록"; 
};

// [개념] 이벤트 직접 연결
// 취소 버튼을 클릭하면 폼 초기화 함수(resetForm)가 즉시 실행되도록 등록합니다.
cancelBtn.addEventListener("click", resetForm);


// ==============================================================================
// 9. 프로그램 실행 시작점 (초기 실행)
// ==============================================================================

// [개념] 최초 함수 호출
// 브라우저가 js 파일을 전부 다 읽고 로딩을 마쳤을 때, 화면에 가장 먼저 데이터 목록을 띄우기 위해 호출합니다.
loadReservations();
