SEQUENCE 생성 예제

CREATE SEQUENCE seq_member
START WITH 1
INCREMENT BY 1;

오라클 시퀀스 객체를 생성

INSERT INTO members(member_id, member_name)
VALUES(seq_member.NEXTVAL, '홍길동');
데이터 추가 시 .nextVAL을 사용해서 적용시킨다.

IN OUT Parameter 예제
CREATE OR REPLACE PROCEDURE increase_number(
    p_num IN OUT NUMBER
)
IS
BEGIN
    p_num := p_num + 1;
END;
/
VARIABLE v_num NUMBER; -- 바인드 변수 생성

EXEC :v_num := 10; -- 바인드 변수에 값을 대입

EXEC increase_number(:v_num);

PRINT v_num;
